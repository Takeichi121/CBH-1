import cron from "node-cron";
import { getAlohaSalesRaw } from "./aloha-service";
import { getNBOSalesAuto } from "./nbo-service"; // 👈 อย่าลืมสร้างไฟล์ nbo-service ที่ใช้รหัส bk1040
import { pushToBoss } from "../socket";
import { streamLLM } from "../replit_integrations/chat/services/llm-router";

export function initProactiveChann() {
  cron.schedule("0 8 * * *", async () => {
    console.log("🌞 Chann ตื่นมาเตรียมรายงานให้เจ้านายแล้วครับ...");

    const alohaData = await getAlohaSalesRaw();
    const nboData = await getNBOSalesAuto(); // 👈 ดึงข้อมูลจาก SQL Server

    let report = "";
    await streamLLM({
      provider: "auto",
      mode: "analysis",
      message: `วิเคราะห์ข้อมูลเช้านี้:
        1. ยอดขาย Aloha (DBF): ${alohaData?.totalFromDbf.toLocaleString() || 0} บาท
        2. ยอดขาย NBO (SQL): ${nboData?.TotalSales.toLocaleString() || 0} บาท
        ช่วยสรุปภาพรวมและทักทายเจ้านายตอนเช้าด้วยครับ`,
      history: [],
      onToken: (token) => { report += token; },
    });

    pushToBoss("chann-alert", {
      title: "อรุณสวัสดิ์ครับเจ้านาย ☕",
      message: report,
    });
  }, { timezone: "Asia/Bangkok" });
}
