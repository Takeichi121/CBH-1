import { eq, between } from "drizzle-orm";
import { db } from "./db"; // ปรับ path ตามโปรเจกต์ของคุณ
import { channConversations, dailySalesReports, shifts } from "./schema";
import OpenAI from "openai";

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

app.post("/api/chann", async (req, res) => {
  const { messages, username } = req.body;
  
  // ดึงข้อความล่าสุดที่ User พิมพ์มา
  const userMessage = messages[messages.length - 1].content;

  // 1. บันทึกข้อความของ User ลง Database
  await db.insert(channConversations).values({
    username: username || "anonymous",
    role: "user",
    content: userMessage,
  });

  // 2. ดึงข้อมูลบริบทร้าน (Store Context)
  const today = new Date();
  // สมมติการ Query ยอดขายและกะทำงาน (ปรับ logic query ตามจริง)
  // const todaySales = await db.select().from(dailySalesReports).where(eq(dailySalesReports.date, today));
  // const weekShifts = await db.select().from(shifts)...
  
  // สร้าง System Prompt พร้อมบริบทร้าน
  const storeContext = `ข้อมูลอัปเดตวันนี้: ยอดขาย 15,400 บาท, พนักงานเข้ากะ 6 คน`; 
  const systemPrompt = {
    role: "system",
    content: `คุณคือ Project Chan ผู้ช่วย AI ประจำร้าน\n${storeContext}`
  };

  // 3. ตั้งค่า Headers สำหรับ SSE (Streaming)
  res.setHeader("Content-Type", "text/event-stream");
  res.setHeader("Cache-Control", "no-cache");
  res.setHeader("Connection", "keep-alive");

  try {
    const stream = await openai.chat.completions.create({
      model: "gpt-4o",
      messages: [systemPrompt, ...messages],
      stream: true,
    });

    let fullAiResponse = ""; // ตัวแปรเก็บข้อความเต็มเพื่อเซฟลง DB ตอนจบ

    // 4. สตรีมข้อมูลกลับไปที่ Frontend
    for await (const chunk of stream) {
      const content = chunk.choices[0]?.delta?.content || "";
      if (content) {
        fullAiResponse += content;
        res.write(`data: ${JSON.stringify({ content })}\n\n`);
      }
    }

    res.write(`data: [DONE]\n\n`);
    res.end();

    // 5. เมื่อสตรีมจบ บันทึกข้อความของ AI ลง Database
    await db.insert(channConversations).values({
      username: username || "anonymous",
      role: "assistant",
      content: fullAiResponse,
    });

  } catch (error) {
    console.error("Chann AI Error:", error);
    res.write(`data: ${JSON.stringify({ content: "\n\n**ระบบขัดข้อง:** ขออภัยค่ะ ไม่สามารถเชื่อมต่อระบบได้ในขณะนี้" })}\n\n`);
    res.write(`data: [DONE]\n\n`);
    res.end();
  }
});
