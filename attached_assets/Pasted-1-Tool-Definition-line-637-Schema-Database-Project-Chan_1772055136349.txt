TypeScript
const { startDate, endDate, ...otherTargets } = args;

// ฟังก์ชันจำลองการบวกวันทีละ 1 วันโดยอิงจาก String (YYYY-MM-DD)
const getNextDayStr = (currentDateStr: string) => {
  const [year, month, day] = currentDateStr.split('-').map(Number);
  const dateObj = new Date(year, month - 1, day);
  dateObj.setDate(dateObj.getDate() + 1);
  
  const nextYear = dateObj.getFullYear();
  const nextMonth = String(dateObj.getMonth() + 1).padStart(2, '0');
  const nextDay = String(dateObj.getDate()).padStart(2, '0');
  return `${nextYear}-${nextMonth}-${nextDay}`;
};

const payload = [];
let currentDateStr = startDate; // Format คาดหวัง: 'YYYY-MM-DD'

// วนลูปสร้าง payload โดยไม่พึ่งพา Timezone
while (currentDateStr <= endDate) {
  payload.push({
    targetDate: currentDateStr, // เปลี่ยนจาก date เป็น targetDate
    ...otherTargets
  });
  currentDateStr = getNextDayStr(currentDateStr);
}
