import { sql } from "drizzle-orm";
import { pgTable, text, varchar, decimal, integer, timestamp, jsonb } from "drizzle-orm/pg-core";
import { createInsertSchema } from "drizzle-zod";
import { z } from "zod";

export const users = pgTable("users", {
  id: varchar("id").primaryKey().default(sql`gen_random_uuid()`),
  username: text("username").notNull().unique(),
  password: text("password").notNull(),
  name: text("name").notNull(),
  role: text("role").notNull().default('staff'),
});

export const dailySalesReports = pgTable("daily_sales_reports", {
  id: varchar("id").primaryKey().default(sql`gen_random_uuid()`),
  reportDate: text("report_date").notNull(),
  reportBy: text("report_by").notNull(),
  workShift: text("work_shift").notNull(),
  
  // Daily Sales
  dailyTarget: decimal("daily_target", { precision: 10, scale: 2 }).notNull(),
  actualSales: decimal("actual_sales", { precision: 10, scale: 2 }).notNull(),
  transactionCount: integer("transaction_count").notNull(),
  
  // MTD (Month To Date)
  mtdTarget: decimal("mtd_target", { precision: 12, scale: 2 }).default('0'),
  mtdActual: decimal("mtd_actual", { precision: 12, scale: 2 }).default('0'),
  mtdTc: integer("mtd_tc").default(0),
  
  // In Store
  dineIn: decimal("dine_in", { precision: 10, scale: 2 }).notNull().default('0'),
  dineInTc: integer("dine_in_tc").default(0),
  takeAway: decimal("take_away", { precision: 10, scale: 2 }).notNull().default('0'),
  takeAwayTc: integer("take_away_tc").default(0),
  
  // Delivery
  delivery: decimal("delivery", { precision: 10, scale: 2 }).notNull().default('0'),
  deliveryBreakdown: jsonb("delivery_breakdown").$type<{
    grabfood?: number;
    lineman?: number;
    shopee?: number;
    foodpanda?: number;
    gokoo?: number;
    bkapp?: number;
  }>().default({}),
  
  // Performance Metrics
  osat: decimal("osat", { precision: 5, scale: 2 }).default('0'),
  surveyCount: integer("survey_count").default(0),
  voidAmount: decimal("void_amount", { precision: 10, scale: 2 }).default('0'),
  voidCount: integer("void_count").default(0),
  
  // Add-ons
  addCheeseCount: integer("add_cheese_count").default(0),
  addCheesePercent: decimal("add_cheese_percent", { precision: 5, scale: 2 }).default('0'),
  vMealCount: integer("v_meal_count").default(0),
  vMealPercent: decimal("v_meal_percent", { precision: 5, scale: 2 }).default('0'),
  upSizeCount: integer("up_size_count").default(0),
  upSizePercent: decimal("up_size_percent", { precision: 5, scale: 2 }).default('0'),
  
  // Waste - Daily
  wasteRawDaily: decimal("waste_raw_daily", { precision: 10, scale: 2 }).default('0'),
  wasteRawDailyPercent: decimal("waste_raw_daily_percent", { precision: 5, scale: 2 }).default('0'),
  wasteMealDaily: decimal("waste_meal_daily", { precision: 10, scale: 2 }).default('0'),
  wasteMealDailyPercent: decimal("waste_meal_daily_percent", { precision: 5, scale: 2 }).default('0'),
  wasteTotalDaily: decimal("waste_total_daily", { precision: 10, scale: 2 }).default('0'),
  wasteTotalDailyPercent: decimal("waste_total_daily_percent", { precision: 5, scale: 2 }).default('0'),
  
  // Waste - MTD
  wasteRawMtd: decimal("waste_raw_mtd", { precision: 10, scale: 2 }).default('0'),
  wasteRawMtdPercent: decimal("waste_raw_mtd_percent", { precision: 5, scale: 2 }).default('0'),
  wasteMealMtd: decimal("waste_meal_mtd", { precision: 10, scale: 2 }).default('0'),
  wasteMealMtdPercent: decimal("waste_meal_mtd_percent", { precision: 5, scale: 2 }).default('0'),
  wasteTotalMtd: decimal("waste_total_mtd", { precision: 10, scale: 2 }).default('0'),
  wasteTotalMtdPercent: decimal("waste_total_mtd_percent", { precision: 5, scale: 2 }).default('0'),
  
  // Labor
  colPercent: decimal("col_percent", { precision: 5, scale: 2 }).default('0'),
  laborHour: decimal("labor_hour", { precision: 6, scale: 2 }).default('0'),
  tcmh: decimal("tcmh", { precision: 5, scale: 2 }).default('0'),
  
  // Manager Roster
  managerRosterDate: text("manager_roster_date"),
  managerRoster: jsonb("manager_roster").$type<Array<{
    name: string;
    group: string;
    shift: string;
    time: string;
  }>>().default([]),
  
  // Staff Roster  
  staffRoster: jsonb("staff_roster").$type<Array<{
    group: string;
    shift: string;
    time: string;
    names: string;
  }>>().default([]),
  
  // Legacy fields for backwards compatibility
  sos: integer("sos"),
  mtdSos: integer("mtd_sos"),
  colAc: integer("col_ac"),
  gsi: decimal("gsi", { precision: 3, scale: 2 }),
  googleReview: decimal("google_review", { precision: 3, scale: 2 }),
  deliveryRating: decimal("delivery_rating", { precision: 3, scale: 2 }),
  wasteDaily: decimal("waste_daily", { precision: 10, scale: 2 }).default('0'),
  mealDaily: decimal("meal_daily", { precision: 10, scale: 2 }).default('0'),
  addCheese: decimal("add_cheese", { precision: 10, scale: 2 }).default('0'),
  vMeal: decimal("v_meal", { precision: 10, scale: 2 }).default('0'),
  upSize: decimal("up_size", { precision: 10, scale: 2 }).default('0'),
  rosterStaff: jsonb("roster_staff").$type<{
    phongsathorn?: string;
    nuttarika?: string;
    chanon?: string;
    wachiraphat?: string;
  }>().default({}),
  
  createdAt: timestamp("created_at").default(sql`NOW()`),
  updatedAt: timestamp("updated_at").default(sql`NOW()`),
});

export const storeSettings = pgTable("store_settings", {
  id: varchar("id").primaryKey().default(sql`gen_random_uuid()`),
  storeName: text("store_name").notNull(),
  storeCode: text("store_code").notNull().unique(),
  dailyTarget: decimal("daily_target", { precision: 10, scale: 2 }).notNull(),
  createdAt: timestamp("created_at").default(sql`NOW()`),
  updatedAt: timestamp("updated_at").default(sql`NOW()`),
});

export const insertUserSchema = createInsertSchema(users).pick({
  username: true,
  password: true,
  name: true,
  role: true,
});

export const insertDailySalesSchema = createInsertSchema(dailySalesReports).omit({
  id: true,
  createdAt: true,
  updatedAt: true,
});

export const insertStoreSettingsSchema = createInsertSchema(storeSettings).omit({
  id: true,
  createdAt: true,
  updatedAt: true,
});

export type InsertUser = z.infer<typeof insertUserSchema>;
export type User = typeof users.$inferSelect;
export type InsertDailySales = z.infer<typeof insertDailySalesSchema>;
export type DailySalesReport = typeof dailySalesReports.$inferSelect;
export type InsertStoreSettings = z.infer<typeof insertStoreSettingsSchema>;
export type StoreSettings = typeof storeSettings.$inferSelect;
