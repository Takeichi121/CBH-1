import { pgTable, serial, text, timestamp } from "drizzle-orm/pg-core";

// ... (Schema เดิมที่มีอยู่ เช่น users, daily_sales_reports, shifts) ...

export const channConversations = pgTable("chann_conversations", {
  id: serial("id").primaryKey(),
  username: text("username").notNull(),
  role: text("role").notNull(), // 'user', 'assistant', หรือ 'system'
  content: text("content").notNull(),
  imageUrl: text("image_url"),
  createdAt: timestamp("created_at").defaultNow().notNull(),
});

export type ChannConversation = typeof channConversations.$inferSelect;
export type InsertChannConversation = typeof channConversations.$inferInsert;
3. ไฟล์ server/routes.ts
ปรับปรุง Endpoint ทั้งหมด ดึง Context ข้อมูลร้าน และประวัติแชทก่อนส่งเข้า OpenAI

TypeScript
import { db } from "../db";
import { channConversations, dailySalesReports, shifts, users } from "../shared/schema";
import { eq, desc, sql } from "drizzle-orm";
import OpenAI from "openai";

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

export function registerRoutes(app: Express) {
  
  // A. Endpoint หลัก (Streaming + Context)
  app.post("/api/chann", async (req, res) => {
    const { message, imageBase64, token } = req.body;
    // สมมติฟังก์ชันหา username (ปรับแก้ตามระบบ Auth ของคุณ)
    const username = req.session?.username || "Ake"; 

    res.setHeader("Content-Type", "text/event-stream");
    res.setHeader("Cache-Control", "no-cache");
    res.setHeader("Connection", "keep-alive");

    try {
      // 1. บันทึกข้อความ User ลง DB
      await db.insert(channConversations).values({
        username,
        role: "user",
        content: message,
        imageUrl: imageBase64 || null,
      });

      // 2. ดึงข้อมูลร้านจาก DB (ปรับ query ตามฟิลด์จริงของคุณ)
      const latestSales = await db.select().from(dailySalesReports).orderBy(desc(dailySalesReports.createdAt)).limit(1);
      const weeklyShifts = await db.select().from(shifts).limit(50); // สมมติดึงงานล่าสุด
      const totalUsersResult = await db.select({ count: sql<number>`count(*)` }).from(users);
      const totalUsers = totalUsersResult[0]?.count || 0;

      // 3. จัดเตรียม System Prompt
      const systemPrompt = `คุณคือ Chann ผู้ช่วย AI ประจำร้าน Burger King สาขา Grand Diamond คอยช่วยเหลือและให้ข้อมูลกับคุณเอก (Shift Manager)
ข้อมูลร้านปัจจุบัน:
- ยอดขายล่าสุด: ${JSON.stringify(latestSales)}
- ตารางงานสัปดาห์นี้: ${JSON.stringify(weeklyShifts)}
- จำนวนพนักงานทั้งหมด: ${totalUsers} คน`;

      const aiMessages: any[] = [{ role: "system", content: systemPrompt }];

      // 4. ดึงประวัติแชท 10 ข้อความล่าสุดมาเป็น Context
      const recentHistory = await db.query.channConversations.findMany({
        where: eq(channConversations.username, username),
        orderBy: [desc(channConversations.createdAt)],
        limit: 10,
      });
      recentHistory.reverse().forEach((msg) => {
        aiMessages.push({ role: msg.role, content: msg.content });
      });

      // (ข้ามการ push ข้อความล่าสุดซ้ำ เพราะอยู่ใน recentHistory แล้ว หรือถ้ายังไม่รวม ให้จัด format ให้รองรับ imageBase64 ถ้ามี)
      if (imageBase64) {
        // แทนที่ข้อความสุดท้ายด้วย format vision
        aiMessages[aiMessages.length - 1] = {
          role: "user",
          content: [
            { type: "text", text: message },
            { type: "image_url", image_url: { url: imageBase64 } }
          ]
        };
      }

      // 5. เรียก OpenAI พร้อม Stream
      const stream = await openai.chat.completions.create({
        model: "gpt-4o",
        messages: aiMessages,
        stream: true,
      });

      let fullAiResponse = "";

      for await (const chunk of stream) {
        const content = chunk.choices[0]?.delta?.content || "";
        if (content) {
          fullAiResponse += content;
          res.write(`data: ${JSON.stringify({ content })}\n\n`);
        }
      }

      // 6. บันทึกข้อความ AI ลง DB ตอนจบ
      await db.insert(channConversations).values({
        username,
        role: "assistant",
        content: fullAiResponse,
      });

      res.write(`data: [DONE]\n\n`);
      res.end();

    } catch (error) {
      console.error("Stream Error:", error);
      res.write(`data: ${JSON.stringify({ content: "\n\n[เกิดข้อผิดพลาดในการเชื่อมต่อ]" })}\n\n`);
      res.write(`data: [DONE]\n\n`);
      res.end();
    }
  });

  // B. Endpoint ดึงประวัติแชท
  app.post("/api/chann/history", async (req, res) => {
    try {
      const username = req.session?.username || "Ake";
      const history = await db.query.channConversations.findMany({
        where: eq(channConversations.username, username),
        orderBy: [desc(channConversations.createdAt)],
        limit: 50,
      });
      res.json({ ok: true, messages: history.reverse() });
    } catch (error) {
      res.status(500).json({ ok: false, error: "Failed to fetch history" });
    }
  });

  // C. Endpoint ล้างประวัติแชท
  app.post("/api/chann/clear", async (req, res) => {
    try {
      const username = req.session?.username || "Ake";
      await db.delete(channConversations).where(eq(channConversations.username, username));
      res.json({ ok: true });
    } catch (error) {
      res.status(500).json({ ok: false, error: "Failed to clear history" });
    }
  });
}
4. ไฟล์ client/src/components/floating-chann-chat.tsx
การแสดงผลฝั่ง Client พร้อม Animation, Markdown และการจัดการ Chunk จาก SSE อย่างถูกต้อง

TypeScript
import React, { useState, useEffect, useRef } from "react";
import ReactMarkdown from "react-markdown";

export function FloatingChannChat({ token }: { token?: string }) {
  const [messages, setMessages] = useState<any[]>([]);
  const [input, setInput] = useState("");
  const [isTyping, setIsTyping] = useState(false);
  const messagesEndRef = useRef<HTMLDivElement>(null);

  const scrollToBottom = () => {
    messagesEndRef.current?.scrollIntoView({ behavior: "smooth" });
  };

  useEffect(() => {
    scrollToBottom();
  }, [messages, isTyping]);

  // B. โหลดประวัติตอนเปิดแชท
  useEffect(() => {
    const fetchHistory = async () => {
      try {
        const res = await fetch("/api/chann/history", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ token }),
        });
        const data = await res.json();
        if (data.ok && data.messages.length > 0) {
          setMessages(data.messages);
        }
      } catch (error) {
        console.error("Failed to load history:", error);
      }
    };
    fetchHistory();
  }, [token]);

  // C. ฟังก์ชันส่งข้อความ
  const sendMessage = async () => {
    if (!input.trim()) return;

    const currentInput = input;
    setInput("");
    
    // เพิ่มข้อความ user และสร้างข้อความ AI เปล่ารอไว้
    setMessages((prev) => [
      ...prev,
      { role: "user", content: currentInput },
      { role: "assistant", content: "" }
    ]);
    setIsTyping(true);

    try {
      const res = await fetch("/api/chann", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ message: currentInput, token }),
      });

      if (!res.body) throw new Error("No response body");

      const reader = res.body.getReader();
      const decoder = new TextDecoder("utf-8");
      let done = false;
      let hasStartedTyping = false;
      let buffer = ""; // เผื่อ chunk ถูกตัดครึ่งบรรทัด

      while (!done) {
        const { value, done: readerDone } = await reader.read();
        done = readerDone;

        if (value) {
          buffer += decoder.decode(value, { stream: true });
          const lines = buffer.split("\n\n");
          buffer = lines.pop() || ""; // เก็บส่วนที่ยังไม่สมบูรณ์ไว้ในรอบถัดไป

          for (const line of lines) {
            if (line.startsWith("data: ")) {
              const dataStr = line.replace("data: ", "");
              
              if (dataStr.trim() === "[DONE]") {
                setIsTyping(false);
                break;
              }

              try {
                const parsed = JSON.parse(dataStr);
                if (parsed.content) {
                  if (!hasStartedTyping) {
                    setIsTyping(false); // ปิด indicator เมื่อ AI เริ่มส่งข้อความมา
                    hasStartedTyping = true;
                  }

                  setMessages((prev) => {
                    const newMsgs = [...prev];
                    const lastIndex = newMsgs.length - 1;
                    newMsgs[lastIndex] = {
                      ...newMsgs[lastIndex],
                      content: newMsgs[lastIndex].content + parsed.content,
                    };
                    return newMsgs;
                  });
                }
              } catch (e) {
                // Ignore parse error สำหรับก้อนยิบย่อย
              }
            }
          }
        }
      }
    } catch (error) {
      console.error("Chat error:", error);
      setIsTyping(false);
    }
  };

  // G. ล้างประวัติแชททั้งหน้าจอและ DB
  const clearHistory = async () => {
    try {
      setMessages([]);
      await fetch("/api/chann/clear", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ token }),
      });
    } catch (error) {
      console.error("Failed to clear history:", error);
    }
  };

  return (
    <div className="flex flex-col h-[600px] w-full max-w-md bg-white dark:bg-zinc-900 border rounded-xl shadow-lg overflow-hidden">
      {/* Header */}
      <div className="flex justify-between items-center p-4 border-b bg-zinc-50 dark:bg-zinc-800">
        <h3 className="font-semibold text-zinc-800 dark:text-zinc-100">Chann Chat</h3>
        <button 
          onClick={clearHistory} 
          className="text-red-500 hover:text-red-600 text-sm font-medium transition-colors"
        >
          ล้างแชท
        </button>
      </div>

      {/* พื้นที่แชท */}
      <div className="flex-1 overflow-y-auto p-4 space-y-4">
        {messages.map((msg, idx) => (
          <div 
            key={idx} 
            // E. Animation
            className={`flex animate-in fade-in slide-in-from-bottom-2 duration-300 ${
              msg.role === "user" ? "justify-end" : "justify-start"
            }`}
          >
            <div className={`p-3 rounded-2xl max-w-[85%] ${
              msg.role === "user" 
                ? "bg-blue-600 text-white rounded-br-sm" 
                : "bg-zinc-100 dark:bg-zinc-800 text-zinc-900 dark:text-zinc-100 rounded-bl-sm"
            }`}>
              {/* D. Markdown rendering */}
              {msg.role === "assistant" ? (
                <ReactMarkdown className="prose prose-sm dark:prose-invert max-w-none break-words leading-relaxed">
                  {msg.content}
                </ReactMarkdown>
              ) : (
                <p className="whitespace-pre-wrap leading-relaxed">{msg.content}</p>
              )}
            </div>
          </div>
        ))}

        {/* F. Typing Indicator ใหม่ */}
        {isTyping && (
          <div className="flex items-center gap-2 text-sm text-zinc-500 animate-in fade-in slide-in-from-bottom-2 duration-300 p-2">
            <span className="relative flex h-2.5 w-2.5">
              <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-blue-400 opacity-75"></span>
              <span className="relative inline-flex rounded-full h-2.5 w-2.5 bg-blue-500"></span>
            </span>
            <span className="animate-pulse">Chann กำลังพิมพ์...</span>
          </div>
        )}
        <div ref={messagesEndRef} />
      </div>

      {/* ช่องป้อนข้อความ */}
      <div className="p-3 border-t bg-white dark:bg-zinc-900">
        <div className="flex gap-2">
          <input 
            type="text" 
            value={input} 
            onChange={(e) => setInput(e.target.value)}
            onKeyDown={(e) => e.key === "Enter" && sendMessage()}
            className="flex-1 border dark:border-zinc-700 bg-transparent rounded-full px-4 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500"
            placeholder="พิมพ์ข้อความถาม Chann..."
          />
          <button 
            onClick={sendMessage} 
            disabled={!input.trim()}
            className="bg-blue-600 hover:bg-blue-700 disabled:opacity-50 text-white px-4 py-2 rounded-full transition-colors font-medium text-sm"
          >
            ส่ง
          </button>
        </div>
      </div>
    </div>
  );
}
