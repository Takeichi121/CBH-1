import OpenAI from "openai";

const openai = new OpenAI({
  apiKey: process.env.AI_INTEGRATIONS_OPENAI_API_KEY,
  baseURL: process.env.AI_INTEGRATIONS_OPENAI_BASE_URL,
});

// ... (โค้ดส่วนอื่นคงเดิม)

// 2. ปรับปรุง Endpoint /api/chann
app.post("/api/chann", async (req, res) => {
  try {
    const { token, message, imageBase64, pageContext } = req.body;
    if (!token || (!message && !imageBase64)) {
      return res.json({ ok: false, message: "Token and message required" });
    }

    // ⚡ OPTIMIZATION: ดึงข้อมูลที่จำเป็นพร้อมกัน (Parallel)
    const [session, tableList] = await Promise.all([
      storage.getSession(token),
      storage.getTableList() // เก็บไว้ข้างนอกหรือ Cache ไว้จะเร็วกว่านี้มาก
    ]);

    if (!session) return res.json({ ok: false, message: "Invalid session" });

    const user = await storage.getUser(session.username);
    if (!user) return res.json({ ok: false, message: "User not found" });

    const username = session.username;
    const isAdmin = user.role === "admin";
    const isManagerOrAdmin = isManagerLike(user.role);

    // ตั้งค่า SSE ทันทีเพื่อลด Time to First Byte (TTFB)
    res.setHeader("Content-Type", "text/event-stream");
    res.setHeader("Cache-Control", "no-cache");
    res.setHeader("Connection", "keep-alive");
    res.setHeader("X-Accel-Buffering", "no");
    res.flushHeaders();

    const userContent = typeof message === "string" ? message.slice(0, 2000) : "";

    // บันทึก Log ลง DB แบบไม่รอ (Non-blocking) เพื่อความเร็ว
    db.insert(channConversations).values({
      username,
      role: "user",
      content: userContent,
      imageUrl: imageBase64 ? "(image attached)" : null,
    }).catch(console.error);

    // ⚡ OPTIMIZATION: ลดขนาด System Prompt 
    // ไม่ใส่ Snapshot ทั้งหมดถ้าไม่จำเป็น หรือส่งแค่ชื่อตาราง
    const systemPrompt = `คุณคือ "Chann" AI อัจฉริยะของ BK Grand Diamond
[Context]
- เวลา: ${nowBangkok().toLocaleTimeString("th-TH")}
- นาย: ${user.nickName || user.fullName} (${user.role})
- ตารางที่เข้าถึงได้: ${tableList.map(t => t.name).join(", ")}
${pageContext ? `\n[Page Context]: ${pageContext}` : ""}
...`; // (บุคลิกภาพอื่นๆ ย่อให้กระชับขึ้น)

    const aiMessages: any[] = [{ role: "system", content: systemPrompt }];

    // ⚡ OPTIMIZATION: ดึงประวัติแชทเฉพาะที่จำเป็น
    const recentHistory = await db.select().from(channConversations)
      .where(eq(channConversations.username, username))
      .orderBy(desc(channConversations.createdAt))
      .limit(10); // ลดจาก 20 เหลือ 10 เพื่อความเร็ว

    recentHistory.reverse().forEach((msg) => {
      if (msg.role === "user" || msg.role === "assistant") {
        aiMessages.push({ role: msg.role, content: msg.content });
      }
    });

    if (imageBase64) {
       // จัดการรูปภาพ (คงเดิม)
    }

    // ⚡ OPTIMIZATION: ใช้ gpt-4o-mini สำหรับ Tool Selection (เร็วขึ้น 2-3 เท่า)
    const TOOL_MODEL = "gpt-4o-mini"; 
    const FINAL_MODEL = "gpt-4o";

    let rounds = 0;
    while (rounds < 3) { // ลดรอบสูงสุดจาก 5 เหลือ 3
      rounds++;
      const response = await openai.chat.completions.create({
        model: TOOL_MODEL,
        messages: aiMessages,
        tools: channTools, // แนะนำให้กรอง Tools ตาม Role ไว้ล่วงหน้าข้างนอก
        tool_choice: "auto",
      });

      const choice = response.choices[0];
      const toolCalls = choice?.message?.tool_calls;

      if (toolCalls && toolCalls.length > 0) {
        aiMessages.push(choice.message);
        // รัน Tool Calls พร้อมกัน (Parallel Tool Execution)
        const toolResults = await Promise.all(toolCalls.map(async (toolCall) => {
          const result = await handleToolCall(toolCall.function.name, JSON.parse(toolCall.function.arguments));
          return {
            role: "tool" as const,
            tool_call_id: toolCall.id,
            content: result,
          };
        }));
        aiMessages.push(...toolResults);
      } else {
        break;
      }
    }

    // ⚡ OPTIMIZATION: Stream ผลลัพธ์สุดท้าย
    const stream = await openai.chat.completions.create({
      model: rounds > 1 ? FINAL_MODEL : TOOL_MODEL, // ถ้ามีการใช้ Tool ให้ใช้ Model ฉลาดสรุป
      messages: aiMessages,
      stream: true,
    });

    let fullAiResponse = "";
    for await (const chunk of stream) {
      const content = chunk.choices[0]?.delta?.content || "";
      if (content) {
        fullAiResponse += content;
        res.write(`data: ${JSON.stringify({ content })}\n\n`);
      }
    }

    // บันทึกคำตอบลง DB (Non-blocking)
    if (fullAiResponse) {
      db.insert(channConversations).values({
        username,
        role: "assistant",
        content: fullAiResponse,
      }).catch(console.error);
    }

    res.write(`data: [DONE]\n\n`);
    res.end();

  } catch (e: any) {
    // Error handling (คงเดิม)
  }
});
