// Route: ดึงค่า Settings มาแสดง
app.post("/api/settings/get-labor", async (req, res) => {
  const settings = await db.select().from(storeSettings).limit(1);
  // ถ้ายังไม่มีค่า ให้ส่งค่า Default กลับไป
  res.json({ 
    ok: true, 
    settings: settings[0] || { 
      rosterHours: 88, 
      dutyDailyHours: 40, 
      ptWageRate: 45, 
      fixedCostDaily: 0, 
      closeShiftDailyCost: 0 
    } 
  });
});

// Route: บันทึกค่า Settings
app.post("/api/settings/save-labor", async (req, res) => {
  const { rosterHours, dutyDailyHours, ptWageRate, fixedCostDaily, closeShiftDailyCost } = req.body;
  
  await db.insert(storeSettings)
    .values({ 
      id: 1, 
      rosterHours, dutyDailyHours, ptWageRate, fixedCostDaily, closeShiftDailyCost, 
      updatedAt: new Date().toISOString() 
    })
    .onConflictDoUpdate({ 
      target: storeSettings.id, 
      set: { 
        rosterHours, dutyDailyHours, ptWageRate, fixedCostDaily, closeShiftDailyCost, 
        updatedAt: new Date().toISOString() 
      } 
    });
  res.json({ ok: true });
});

// Utility: ฟังก์ชันคำนวณสูตร (Logic หัวใจสำคัญ)
async function calculateLaborLogic(date: string, inputs: any) {
  // 1. ดึงค่า Settings
  const settingRes = await db.select().from(storeSettings).limit(1);
  const cfg = settingRes[0] || { rosterHours: 88, dutyDailyHours: 40, ptWageRate: 45, fixedCostDaily: 0, closeShiftDailyCost: 0 };
  
  // 2. ดึงยอดขายและ TC ของวันนั้น
  const salesData = await storage.getDailySalesReportByDate(date);
  const sales = Number(salesData?.salesActual || 0);
  const tc = Number(salesData?.transactionCount || 0);

  // 3. ตัวแปรรายวัน (Input)
  const actual = Number(inputs.actualHours || 0); // PT+FT
  const ot = Number(inputs.otHours || 0);

  // --- เริ่มคำนวณ ---
  
  // A. Summary Hours = Duty (Fix) + Actual + OT
  const summaryHours = (cfg.dutyDailyHours || 40) + actual + ot;

  // B. Variance = Summary - Roster
  const varianceHours = summaryHours - (cfg.rosterHours || 88);

  // C. Labor Cost (Baht)
  // สูตร: (Fix Cost รายวัน + ค่ารถปิดร้าน) + ((Actual + OT) * เรทค่าแรง PT)
  // *หมายเหตุ: ตรงนี้สมมติว่าชั่วโมง Actual+OT จ่ายด้วยเรท PT (หรือปรับสูตรตามจริงได้)
  const variableCost = (actual + ot) * (cfg.ptWageRate || 0);
  const fixedCost = (cfg.fixedCostDaily || 0) + (cfg.closeShiftDailyCost || 0);
  const laborCostTotal = fixedCost + variableCost;

  // D. % COL = Cost / Sales * 100
  const colPercent = sales > 0 ? (laborCostTotal / sales) * 100 : 0;

  // E. TCMH = TC / Summary Hours
  const tcmh = summaryHours > 0 ? (tc / summaryHours) : 0;

  return {
    ...inputs,
    summaryHours,
    varianceHours,
    laborCostTotal,
    colPercent,
    tcmh
  };
}

// Route: บันทึกข้อมูลรายวัน
app.post("/api/sales/save-daily-labor", async (req, res) => {
  const { date, actualHours, otHours } = req.body;
  try {
    const result = await calculateLaborLogic(date, { actualHours, otHours });
    
    // บันทึกลง DB
    await db.insert(dailyLabor)
      .values({ date, ...result })
      .onConflictDoUpdate({ target: dailyLabor.date, set: result });

    res.json({ ok: true, data: result });
  } catch (e: any) {
    res.json({ ok: false, message: e.message });
  }
});
