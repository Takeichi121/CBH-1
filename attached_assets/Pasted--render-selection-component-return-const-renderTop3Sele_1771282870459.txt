// ฟังก์ชัน render แถว selection (วางไว้ใน component หลัก ก่อน return)
  const renderTop3Selectors = (
    type: 'waste' | 'unac',
    selections: typeof wasteSelections
  ) => {
    return (
      <div className="space-y-2 border rounded-md p-3 bg-muted/20">
        {selections.map((row, index) => (
          <div key={index} className="flex items-center gap-2">
            <span className="text-sm font-medium w-4">{index + 1}.</span>
            
            {/* Dropdown เลือก Item */}
            <Popover
              open={openCombobox?.type === type && openCombobox?.index === index}
              onOpenChange={(isOpen) => 
                setOpenCombobox(isOpen ? { type, index } : null)
              }
            >
              <PopoverTrigger asChild>
                <Button
                  variant="outline"
                  role="combobox"
                  className={cn(
                    "w-[200px] justify-between",
                    !row.itemName && "text-muted-foreground"
                  )}
                >
                  {row.itemName || "เลือก Item..."}
                  <ChevronsUpDown className="ml-2 h-4 w-4 shrink-0 opacity-50" />
                </Button>
              </PopoverTrigger>
              <PopoverContent className="w-[200px] p-0">
                <Command>
                  <CommandInput placeholder="ค้นหา item..." />
                  <CommandList>
                    <CommandEmpty>ไม่พบข้อมูล</CommandEmpty>
                    <CommandGroup>
                      {borrowItems.map((item) => (
                        <CommandItem
                          key={item.id}
                          value={item.name}
                          onSelect={(currentValue) => {
                            // ถ้าเลือกซ้ำให้ยกเลิก หรือถ้าเลือกใหม่ให้เปลี่ยน
                            updateSelection(type, index, 'itemName', currentValue === row.itemName ? "" : currentValue);
                            setOpenCombobox(null);
                          }}
                        >
                          <Check
                            className={cn(
                              "mr-2 h-4 w-4",
                              row.itemName === item.name ? "opacity-100" : "opacity-0"
                            )}
                          />
                          {item.name}
                        </CommandItem>
                      ))}
                    </CommandGroup>
                  </CommandList>
                </Command>
              </PopoverContent>
            </Popover>

            {/* ช่องกรอกรายละเอียด */}
            <Input
              placeholder="รายละเอียด (เช่น 500 บาท)"
              value={row.detail}
              onChange={(e) => updateSelection(type, index, 'detail', e.target.value)}
              className="flex-1"
            />
            
            {/* ปุ่ม Clear แถว */}
            {(row.itemName || row.detail) && (
                <Button 
                    variant="ghost" 
                    size="icon" 
                    onClick={() => {
                        updateSelection(type, index, 'itemName', '');
                        updateSelection(type, index, 'detail', '');
                    }}
                >
                    <X className="h-4 w-4 text-muted-foreground" />
                </Button>
            )}
          </div>
        ))}
      </div>
    );
  };
