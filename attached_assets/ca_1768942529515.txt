import { Button } from "@/components/ui/button";
import { CalendarPlus } from "lucide-react";

interface ShiftProps {
  title: string; // เช่น "กะเช้า (Morning Shift)"
  startTime: Date;
  endTime: Date;
}

export function AddToCalendarBtn({ title, startTime, endTime }: ShiftProps) {
  const handleAddToCalendar = () => {
    // แปลงวันที่เป็น format ที่ Google ต้องการ (YYYYMMDDTHHmmssZ)
    const formatTime = (date: Date) => date.toISOString().replace(/-|:|\.\d\d\d/g, "");
    
    const start = formatTime(startTime);
    const end = formatTime(endTime);
    
    // สร้าง URL
    const googleUrl = `https://calendar.google.com/calendar/render?action=TEMPLATE&text=${encodeURIComponent(title)}&dates=${start}/${end}&details=งานที่+Burger+King+Grand+Diamond&location=Burger+King+Grand+Diamond`;

    window.open(googleUrl, '_blank');
  };

  return (
    <Button variant="outline" size="sm" onClick={handleAddToCalendar}>
      <CalendarPlus className="mr-2 h-4 w-4" />
      Add to Calendar
    </Button>
  );
}
