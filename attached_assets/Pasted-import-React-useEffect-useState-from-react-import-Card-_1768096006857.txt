import React, { useEffect, useState } from 'react';
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Input } from "@/components/ui/input";
import { Button } from "@/components/ui/button";
import { Label } from "@/components/ui/label";
import { useToast } from "@/components/ui/use-toast";

export default function LaborSettings() {
  const { toast } = useToast();
  const [config, setConfig] = useState({
    rosterHours: 88,        // Roster Area
    dutyDailyHours: 40,     // Duty Team (Fix)
    ptWageRate: 45,         // ค่าแรง PT
    fixedCostDaily: 0,      // เงินเดือน FT/Manager (ต่อวัน)
    closeShiftDailyCost: 0, // ค่ารถปิดร้าน (ต่อวัน)
  });

  useEffect(() => {
    // โหลดค่าเดิม
    fetch('/api/settings/get-labor', { method: 'POST' })
      .then(res => res.json())
      .then(data => { if(data.ok) setConfig(prev => ({...prev, ...data.settings})); });
  }, []);

  const handleSave = async () => {
    const res = await fetch('/api/settings/save-labor', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(config)
    });
    if (res.ok) toast({ title: "บันทึกสำเร็จ", description: "อัปเดตค่าคงที่แรงงานแล้ว" });
  };

  return (
    <Card className="max-w-xl mx-auto mt-6 shadow-md">
      <CardHeader className="bg-blue-50 border-b">
        <CardTitle className="text-blue-800">⚙️ ตั้งค่าค่าคงที่แรงงาน (Labor Constants)</CardTitle>
      </CardHeader>
      <CardContent className="space-y-6 pt-6">
        
        {/* กลุ่ม 1: ชั่วโมง */}
        <div className="grid grid-cols-2 gap-6">
          <div className="space-y-2">
            <Label>Roster Commit (ชม./วัน)</Label>
            <Input type="number" value={config.rosterHours} 
              onChange={e => setConfig({...config, rosterHours: parseFloat(e.target.value)})} />
            <p className="text-xs text-gray-500">ค่าจาก Area (เช่น 88, 96)</p>
          </div>
          <div className="space-y-2">
            <Label>Duty Team (ชม./วัน)</Label>
            <Input type="number" value={config.dutyDailyHours} 
              onChange={e => setConfig({...config, dutyDailyHours: parseFloat(e.target.value)})} />
            <p className="text-xs text-gray-500">ค่าคงที่ (เช่น 40)</p>
          </div>
        </div>

        <hr className="border-gray-100" />

        {/* กลุ่ม 2: ต้นทุน (บาท) */}
        <div className="space-y-4">
          <div className="grid grid-cols-2 gap-6">
             <div className="space-y-2">
              <Label>ค่าแรง PT (บาท/ชม.)</Label>
              <Input type="number" value={config.ptWageRate} 
                onChange={e => setConfig({...config, ptWageRate: parseFloat(e.target.value)})} />
            </div>
            <div className="space-y-2">
              <Label>ค่ารถปิดร้าน (บาท/วัน)</Label>
              <Input type="number" value={config.closeShiftDailyCost} 
                onChange={e => setConfig({...config, closeShiftDailyCost: parseFloat(e.target.value)})} />
              <p className="text-xs text-gray-500">เหมาจ่ายต่อวัน</p>
            </div>
          </div>

          <div className="space-y-2">
            <Label>เงินเดือนเฉลี่ย FT/Manager (บาท/วัน)</Label>
            <Input type="number" value={config.fixedCostDaily} 
              onChange={e => setConfig({...config, fixedCostDaily: parseFloat(e.target.value)})} />
            <p className="text-xs text-gray-500">Fixed Cost เฉลี่ยรายวัน</p>
          </div>
        </div>

        <Button className="w-full bg-blue-600 hover:bg-blue-700" onClick={handleSave}>
          บันทึกการตั้งค่า
        </Button>
      </CardContent>
    </Card>
  );
}
