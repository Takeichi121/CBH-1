แทนที่ฟังก์ชัน handleCopyReport ด้วยโค้ดด้านล่างนี้ (อัปเดต Template และการคำนวณตามโจทย์)

  const handleCopyReport = () => {
    const v = form.getValues();

    // Sales Values
    const actualSalesVal = parseFloat(v.actualSales) || 0;
    const dailyTargetVal = parseFloat(v.dailyTarget) || 0;
    const mtdActualVal = parseFloat(v.mtdActual) || 0;
    const mtdTargetVal = parseFloat(v.mtdTarget) || 0;

    // TC Values
    const tcVal = parseFloat(v.transactionCount) || 0;
    const mtdTcVal = parseFloat(v.mtdTransactionCount) || 0;

    // TA Values
    const taVal = tcVal > 0 ? (actualSalesVal / tcVal).toFixed(0) : "0";
    const mtdTaVal = mtdTcVal > 0 ? (mtdActualVal / mtdTcVal).toFixed(0) : "0";

    // Channel Values & Percentages
    const dineInVal = parseFloat(v.dineIn) || 0;
    const takeAwayVal = parseFloat(v.takeAway) || 0;
    const deliveryTotal = parseFloat(v.delivery) || 0;

    const dineInPercent = actualSalesVal > 0 ? ((dineInVal / actualSalesVal) * 100).toFixed(0) : "0";
    const takeAwayPercent = actualSalesVal > 0 ? ((takeAwayVal / actualSalesVal) * 100).toFixed(0) : "0";
    const deliveryTotalPercent = actualSalesVal > 0 ? ((deliveryTotal / actualSalesVal) * 100).toFixed(0) : "0";

    // Delivery Platforms (รวม Robin, GoKOO)
    const grabVal = parseFloat(v.grab) || 0;
    const linemanVal = parseFloat(v.lineman) || 0;
    const shopeeVal = parseFloat(v.shopee) || 0;
    const robinVal = parseFloat(v.robin) || 0;
    const gokooVal = parseFloat(v.gokoo) || 0;

    const grabPercent = actualSalesVal > 0 ? ((grabVal / actualSalesVal) * 100).toFixed(0) : "0";
    const linemanPercent = actualSalesVal > 0 ? ((linemanVal / actualSalesVal) * 100).toFixed(0) : "0";
    const shopeePercent = actualSalesVal > 0 ? ((shopeeVal / actualSalesVal) * 100).toFixed(0) : "0";
    const robinPercent = actualSalesVal > 0 ? ((robinVal / actualSalesVal) * 100).toFixed(0) : "0";
    const gokooPercent = actualSalesVal > 0 ? ((gokooVal / actualSalesVal) * 100).toFixed(0) : "0";

    // Waste Logic
    const wasteDailyTotalVal = parseFloat(v.wasteDailyTotal) || 0;
    const wasteMtdTotalVal = parseFloat(v.wasteMtdTotal) || 0;
    
    // คำนวณ Waste % (เทียบกับยอดขาย)
    const wasteDailyPercent = actualSalesVal > 0 ? ((wasteDailyTotalVal / actualSalesVal) * 100).toFixed(2) : "0.00";
    const wasteMtdPercent = mtdActualVal > 0 ? ((wasteMtdTotalVal / mtdActualVal) * 100).toFixed(2) : "0.00";

    // Hours (Task 3: ไม่รวมทีมผู้จัดการ)
    const actualHoursVal = parseFloat(v.actualHours) || 0;
    const otHoursVal = parseFloat(v.otHours) || 0;
    const workHour = (actualHoursVal + otHoursVal).toFixed(1);

    // Roster Logic (Mockup or existing logic)
    const managerRosterLines = "Manager List..."; // ดึงจาก Logic เดิมของคุณ
    const staffRosterLines = "Staff List...";     // ดึงจาก Logic เดิมของคุณ
    const nextDayFormatted = "DD/MM/YYYY";        // ดึงจาก Logic เดิมของคุณ

    // === Report Template (Task 4) ===
    const reportText = `🗓️${format(new Date(v.reportDate), "dd/MM/yyyy")}

💵Daily Sales=${actualSalesVal.toLocaleString()}/${dailyTargetVal.toLocaleString()}
MTD Sale=${mtdActualVal.toLocaleString()}/${mtdTargetVal.toLocaleString()}

👨‍👩‍👧‍👦Daily TC =${tcVal.toLocaleString()}
👨‍👩‍👧‍👦MTD TC =${mtdTcVal.toLocaleString()}
👑 Daily TA =${taVal}
👑 MTD TA =${mtdTaVal}

🍽 Dinein :${dineInVal.toLocaleString()}/${dineInPercent}%
🛍Takeaway :${takeAwayVal.toLocaleString()}/${takeAwayPercent}%
🛵Delivery :${deliveryTotal.toLocaleString()}/${deliveryTotalPercent}%

🛵Grab Food :${grabVal.toLocaleString()}/${grabPercent}%
🛵Line Man :${linemanVal.toLocaleString()}/${linemanPercent}%
🛵 Shopeefood:${shopeeVal.toLocaleString()}/${shopeePercent}%
🛵Robin:${robinPercent}%
🛵 GoKOO:${gokooPercent}%

🏃🏻‍♀️SOS =${v.sosDaily}
🏃🏻MTD SOS =${v.sosMtd}

🗑WasteDaily :${wasteDailyTotalVal.toLocaleString()}/${wasteDailyPercent}%
🗑 WasteMTD:${wasteMtdTotalVal.toLocaleString()}/${wasteMtdPercent}%

🕰Work Hour :${workHour} (ไม่รวมทีมผู้จัดการ)

📝OSAT:${v.osatScore}/${v.osatComments}comments

🗓️S[week] ${nextDayFormatted}📌
Roster Manager,Duty
${managerRosterLines}

📄Roster Tomorrow
${staffRosterLines}`;

    // Copy to clipboard
    navigator.clipboard.writeText(reportText);
    toast({ title: "Copied!", description: "รายงานถูกคัดลอกแล้ว" });
  };
