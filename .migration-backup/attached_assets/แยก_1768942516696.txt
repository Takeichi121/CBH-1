import { useAuth } from "@/hooks/use-auth"; // สมมติว่ามี hook นี้สำหรับดึง user
import { SalesChart } from "@/components/dashboard/SalesChart";
import { DailyTip } from "@/components/dashboard/DailyTip";

export default function Dashboard() {
  const { user } = useAuth();

  if (!user) return <div>Loading...</div>;

  return (
    <div className="flex-1 space-y-4 p-8 pt-6">
      <div className="flex items-center justify-between space-y-2">
        <h2 className="text-3xl font-bold tracking-tight">
            สวัสดี, {user.username} ({user.role})
        </h2>
      </div>

      {/* --- ส่วนของ Manager --- */}
      {user.role === 'manager' || user.role === 'admin' ? (
        <div className="grid gap-4 md:grid-cols-2 lg:grid-cols-7">
          <SalesChart />
          <div className="col-span-3 space-y-4">
             {/* ใส่ Component รายชื่อพนักงานตรงนี้ */}
             <div className="rounded-xl border bg-card p-6">
                <h3 className="font-semibold">พนักงานที่เข้ากะวันนี้</h3>
                {/* map list พนักงาน... */}
             </div>
          </div>
        </div>
      ) : (
        /* --- ส่วนของ Staff --- */
        <div className="grid gap-4 md:grid-cols-2 lg:grid-cols-4">
          <div className="col-span-2">
            <h3 className="text-xl font-bold mb-4">ตารางงานของคุณ</h3>
            {/* ใส่ Component ตารางงาน Staff ตรงนี้ */}
          </div>
          <div className="col-span-2">
             <DailyTip />
          </div>
        </div>
      )}
    </div>
  );
}
