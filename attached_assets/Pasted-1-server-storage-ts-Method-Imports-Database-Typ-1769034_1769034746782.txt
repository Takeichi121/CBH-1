// 1. เช็ค Imports ด้านบนสุด (เพิ่ม eq, and ถ้ายังไม่มี)
import { users, shifts } from "@shared/schema"; // ต้องมี users และ shifts
import { eq, and } from "drizzle-orm"; // เพิ่ม eq และ and

// 2. เพิ่มใน interface IStorage
export interface IStorage {
  // ... methods เดิม ...
  upsertShift(shift: { username: string; date: string; shiftGroup: string }): Promise<any>;
}

// 3. เพิ่มใน class DatabaseStorage
export class DatabaseStorage implements IStorage {
  // ... methods เดิม ...

  // เพิ่ม method นี้ต่อท้าย
  async upsertShift(shift: { username: string; date: string; shiftGroup: string }) {
    // เช็คว่ามีข้อมูลเดิมไหม (User คนเดิม + วันเดิม)
    const existing = await db.select().from(shifts)
      .where(and(eq(shifts.username, shift.username), eq(shifts.date, shift.date)))
      .limit(1);

    if (existing.length > 0) {
      // มีแล้ว -> อัปเดต (Update)
      const [updated] = await db.update(shifts)
        .set({ shiftGroup: shift.shiftGroup })
        .where(and(eq(shifts.username, shift.username), eq(shifts.date, shift.date)))
        .returning();
      return updated;
    } else {
      // ยังไม่มี -> สร้างใหม่ (Insert)
      const [inserted] = await db.insert(shifts)
        .values({
          username: shift.username,
          date: shift.date,
          shiftGroup: shift.shiftGroup,
          createdAt: new Date().toISOString(), // หรือใช้วันที่ปัจจุบัน
        })
        .returning();
      return inserted;
    }
  }
}
