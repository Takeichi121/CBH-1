✅ จุดสำคัญที่ต้องให้ GPT ทำตามแบบเป๊ะ

filter users:

const managerTeam =
  rosterData?.users?.filter((u: any) => u.role === "manager" && u.active === 1) || [];


filter roster shifts ให้เหลือเฉพาะ managers:

const managerShifts =
  rosterData?.roster?.filter((s: any) =>
    managerTeam.some((m: any) => m.username === s.username)
  ) || [];


ถ้าไม่มี shift ให้แสดง message:

{managerShifts.length === 0 ? (...) : (...)}
