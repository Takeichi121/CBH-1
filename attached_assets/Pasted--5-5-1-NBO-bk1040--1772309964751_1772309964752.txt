ไฟล์: server/services/nbo-service.ts

TypeScript
import sql from "mssql";

const nboConfig = {
  user: "bk1040", // หรือ 'sa' ตามที่เจ้านายสืบพบ
  password: "bk1040",
  server: "localhost", // หากรันคนละเครื่องให้ใส่ IP ของเครื่อง NBO
  database: "AlohaNBO", // ชื่อ DB มาตรฐานของระบบ Radiant/NCR
  options: {
    encrypt: false,
    trustServerCertificate: true
  }
};

export async function getNBOSalesAuto() {
  try {
    const pool = await sql.connect(nboConfig);
    const result = await pool.request().query(`
      SELECT 
        ISNULL(SUM(NetSales), 0) as TotalSales, 
        COUNT(CheckID) as GuestCount
      FROM GuestCheck
      WHERE BusinessDate = CAST(GETDATE() AS DATE)
    `);
    await pool.close();
    return result.recordset[0];
  } catch (err) {
    console.error("🔴 [NBO SQL Error]:", err);
    return null;
  }
}
🤖 ปรับปรุงขั้นตอนที่ 5: proactive-agent.ts (ฉบับรวมพลัง)
เราจะปรับให้ Chann ตื่นมาแล้วดูข้อมูลจากทั้ง Aloha (DBF) และ NBO (SQL) พร้อมกันครับ

TypeScript
import cron from "node-cron";
import { getAlohaSalesRaw } from "./aloha-service";
import { getNBOSalesAuto } from "./nbo-service"; // 👈 เพิ่มการนำเข้า
import { pushToBoss } from "../socket";
import { streamLLM } from "../replit_integrations/chat/services/llm-router";

export function initProactiveChann() {
  cron.schedule("0 8 * * *", async () => {
    console.log("🌞 Chann กำลังวิเคราะห์ข้อมูลรวมจาก Aloha และ NBO...");

    const [alohaData, nboData] = await Promise.all([
      getAlohaSalesRaw(),
      getNBOSalesAuto()
    ]);

    let report = "";
    const prompt = `
      เจ้านายครับ นี่คือสรุปยอดเมื่อวาน:
      - ยอดจาก Aloha (DBF): ${alohaData?.totalFromDbf.toLocaleString() || 0} บาท
      - ยอดจาก NBO (SQL): ${nboData?.TotalSales.toLocaleString() || 0} บาท
      ช่วยวิเคราะห์ความสอดคล้องของข้อมูล และเสนอแผนเชิงรุกให้เจ้านาย 3 ข้อครับ
    `;

    await streamLLM({
      provider: "auto",
      mode: "analysis",
      message: prompt,
      history: [],
      onToken: (token) => { report += token; },
    });

    pushToBoss("chann-alert", {
      title: "รายงานเช้านี้จาก Chann 📊",
      message: report,
    });
  }, { timezone: "Asia/Bangkok" });
}
