1. ระบบดึงข้อมูลจาก Aloha (DBF Reader)
เราจะใช้ dbffile เข้าไปอ่านไฟล์ยอดขายรายวัน (GNDSALE.dbf) ซึ่งเป็นมาตรฐานของระบบ Radiant/NCR ครับ

ไฟล์: server/services/aloha-service.ts

TypeScript
import { DBFFile } from 'dbffile';
import path from 'path';

export async function getAlohaSalesRaw() {
  try {
    // 💡 ระบุ Path ไปยังโฟลเดอร์ DATA ของ Aloha (ปรับตามจริงที่ร้านนะครับ)
    const alohaDataPath = path.resolve('C:/Aloha/DATA/GNDSALE.dbf'); 
    
    const dbf = await DBFFile.open(alohaDataPath);
    const records = await dbf.readRecords(100); // ดึง 100 รายการล่าสุดมาวิเคราะห์

    // คำนวณยอดเบื้องต้นจากไฟล์ DBF
    const totalFromDbf = records.reduce((sum, rec: any) => sum + (Number(rec.AMOUNT) || 0), 0);

    return {
      totalFromDbf,
      lastRecordCount: records.length,
      recentItems: records.slice(0, 5) // ส่งตัวอย่าง 5 รายการให้ AI ดู
    };
  } catch (error) {
    console.error("🔴 [Aloha DBF Error]:", error);
    return null;
  }
}
2. อัปเกรด Proactive AI (ดึงทั้ง NBO + Aloha)
เราจะปรับให้ Chann ตื่นมาแล้วเช็คข้อมูลจากทั้ง 2 แหล่ง เพื่อเปรียบเทียบความถูกต้อง (Reconciliation) แบบออโต้ครับ

ไฟล์: server/services/proactive-agent.ts (ฉบับอัปเกรด)

TypeScript
import cron from 'node-cron';
import { getNBOSalesAuto } from "./nbo-service";
import { getAlohaSalesRaw } from "./aloha-service"; // 👈 เพิ่มตัวนี้
import { pushToBoss } from "../socket";
import { streamLLM } from "./llm-router";

export function initProactiveChann() {
  cron.schedule('0 8 * * *', async () => {
    // 1. ดึงข้อมูลจาก SQL Server (NBO) ด้วยรหัส bk1040
    const nboData = await getNBOSalesAuto();
    // 2. ดึงข้อมูลดิบจากไฟล์ Aloha (DBF)
    const alohaData = await getAlohaSalesRaw();

    if (!nboData && !alohaData) return;

    let analysis = "";
    const prompt = `
      ข้อมูลเช้านี้:
      - ยอดจาก NBO (SQL): ${nboData?.TotalSales || 0} บาท
      - ยอดจาก Aloha (DBF): ${alohaData?.totalFromDbf || 0} บาท
      ช่วยวิเคราะห์ความสอดคล้องและสรุปแผนงานเชิงรุกให้เจ้านายทีครับ
    `;

    await streamLLM({
      provider: "auto",
      mode: "analysis",
      message: prompt,
      history: [],
      onToken: (token) => { analysis += token; }
    });

    pushToBoss("chann-alert", {
      title: "Chann Report: Aloha & NBO 📊",
      message: analysis
    });
  });
}
