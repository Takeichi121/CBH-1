อัปเดต Endpoint เดิมและเพิ่ม Endpoint ใหม่สำหรับการจัดการประวัติแชท:

TypeScript
import { channConversations } from "../shared/schema";
import { eq, desc } from "drizzle-orm";
// ... imports อื่นๆ ...

export function registerRoutes(app: Express) {
  
  // 1. Endpoint หลัก (สตรีมมิ่ง)
  app.post("/api/chann", async (req, res) => {
    const { message, imageBase64, token } = req.body;
    
    // สมมติฟังก์ชันหา username จาก session/token
    const username = await getUsernameFromToken(token); 

    // ตั้งค่า Headers สำหรับ SSE
    res.setHeader("Content-Type", "text/event-stream");
    res.setHeader("Cache-Control", "no-cache");
    res.setHeader("Connection", "keep-alive");

    try {
      // บันทึกข้อความ User ลง DB ก่อนเริ่ม Stream
      await db.insert(channConversations).values({
        username,
        role: "user",
        content: message,
        imageUrl: imageBase64 || null,
      });

      // ดึงข้อมูลยอดขายและตารางงาน (คุณสามารถปรับ query ให้ตรงกับตารางจริงได้)
      const todaySales = await db.query.sales.findFirst({ /* query ยอดขายวันนี้ */ });
      const weekSchedule = await db.query.schedules.findMany({ /* query ตารางงานสัปดาห์นี้ */ });

      const systemPrompt = `
        คุณคือ Chann ข้อมูลสำหรับวันนี้:
        - ยอดขายวันนี้: ${JSON.stringify(todaySales || {})}
        - ตารางงานสัปดาห์นี้: ${JSON.stringify(weekSchedule || [])}
      `;

      const messages = [
        { role: "system", content: systemPrompt }
      ];

      // รองรับ Vision (imageBase64)
      if (imageBase64) {
        messages.push({
          role: "user",
          content: [
            { type: "text", text: message },
            { type: "image_url", image_url: { url: imageBase64 } }
          ]
        });
      } else {
        messages.push({ role: "user", content: message });
      }

      const stream = await openai.chat.completions.create({
        model: "gpt-4o", // หรือโมเดลที่รองรับ vision
        messages: messages,
        stream: true,
      });

      let fullAiResponse = "";

      for await (const chunk of stream) {
        const content = chunk.choices[0]?.delta?.content || "";
        if (content) {
          fullAiResponse += content;
          res.write(`data: ${JSON.stringify({ content })}\n\n`);
        }
      }

      // บันทึกข้อความ AI ลง DB หลัง Stream จบ
      await db.insert(channConversations).values({
        username,
        role: "assistant",
        content: fullAiResponse,
      });

      res.write(`data: [DONE]\n\n`);
      res.end();

    } catch (error) {
      console.error("Stream Error:", error);
      res.write(`data: ${JSON.stringify({ error: "Internal Server Error" })}\n\n`);
      res.end();
    }
  });

  // 2. Endpoint ดึงประวัติแชท
  app.post("/api/chann/history", async (req, res) => {
    try {
      const { token } = req.body;
      const username = await getUsernameFromToken(token);

      const history = await db.query.channConversations.findMany({
        where: eq(channConversations.username, username),
        orderBy: [desc(channConversations.createdAt)],
        limit: 50,
      });

      // สลับลำดับให้ข้อความเก่าอยู่บน ข้อความใหม่อยู่ล่าง
      res.json({ ok: true, messages: history.reverse() });
    } catch (error) {
      res.status(500).json({ ok: false, error: "Failed to fetch history" });
    }
  });

  // 3. Endpoint ล้างประวัติแชท
  app.post("/api/chann/clear", async (req, res) => {
    try {
      const { token } = req.body;
      const username = await getUsernameFromToken(token);

      await db.delete(channConversations).where(eq(channConversations.username, username));
      res.json({ ok: true });
    } catch (error) {
      res.status(500).json({ ok: false, error: "Failed to clear history" });
    }
  });
}
ไฟล์ 3: client/src/components/floating-chann-chat.tsx
ปรับปรุงคอมโพเนนต์หลักเพื่อรองรับการสตรีมและ React Markdown:

TypeScript
import React, { useState, useEffect } from "react";
import ReactMarkdown from "react-markdown";
// ... imports อื่นๆ ที่คุณใช้อยู่ (เช่น UI components, lucide-react)

export function FloatingChannChat({ token }) {
  const [messages, setMessages] = useState<any[]>([]);
  const [input, setInput] = useState("");
  const [isTyping, setIsTyping] = useState(false);

  // โหลดประวัติตอนเปิดแชท
  useEffect(() => {
    const fetchHistory = async () => {
      try {
        const res = await fetch("/api/chann/history", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ token }),
        });
        const data = await res.json();
        if (data.ok) {
          setMessages(data.messages);
        }
      } catch (error) {
        console.error("Failed to load history:", error);
      }
    };
    
    fetchHistory();
  }, [token]);

  const sendMessage = async () => {
    if (!input.trim()) return;

    const userMsg = { role: "user", content: input };
    setMessages((prev) => [...prev, userMsg]);
    setInput("");
    setIsTyping(true);

    // สร้างข้อความ AI เปล่ารอไว้ก่อน
    setMessages((prev) => [...prev, { role: "assistant", content: "" }]);

    try {
      const res = await fetch("/api/chann", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ message: input, token /* , imageBase64: ... */ }),
      });

      if (!res.body) throw new Error("No response body");

      const reader = res.body.getReader();
      const decoder = new TextDecoder("utf-8");
      let done = false;
      let hasStartedTyping = false;

      while (!done) {
        const { value, done: readerDone } = await reader.read();
        done = readerDone;

        if (value) {
          const chunk = decoder.decode(value, { stream: true });
          const lines = chunk.split("\n\n"); // แยกตาม SSE format

          for (const line of lines) {
            if (line.startsWith("data: ")) {
              const dataStr = line.replace("data: ", "");
              
              if (dataStr === "[DONE]") {
                setIsTyping(false);
                break;
              }

              try {
                const parsed = JSON.parse(dataStr);
                if (parsed.content) {
                  // ทันทีที่ AI เริ่มตอบ ให้ปิด typing indicator
                  if (!hasStartedTyping) {
                    setIsTyping(false);
                    hasStartedTyping = true;
                  }

                  // อัปเดตเนื้อหาข้อความสุดท้าย (ของ AI)
                  setMessages((prev) => {
                    const newMsgs = [...prev];
                    const lastIndex = newMsgs.length - 1;
                    newMsgs[lastIndex] = {
                      ...newMsgs[lastIndex],
                      content: newMsgs[lastIndex].content + parsed.content,
                    };
                    return newMsgs;
                  });
                }
              } catch (e) {
                // ข้าม error หาก JSON แตกจากการส่งข้าม chunk
              }
            }
          }
        }
      }
    } catch (error) {
      console.error("Send message error:", error);
      setIsTyping(false);
    }
  };

  const clearHistory = async () => {
    try {
      const res = await fetch("/api/chann/clear", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ token }),
      });
      const data = await res.json();
      if (data.ok) {
        setMessages([]);
      }
    } catch (error) {
      console.error("Failed to clear history:", error);
    }
  };

  return (
    <div className="flex flex-col h-full">
      {/* Header และปุ่ม Clear History (ปรับ CSS ตามที่คุณออกแบบไว้) */}
      <div className="flex justify-between p-4 border-b">
        <h3>Chann Chat</h3>
        <button onClick={clearHistory} className="text-red-500 text-sm">เคลียร์แชท</button>
      </div>

      {/* พื้นที่แสดงข้อความ */}
      <div className="flex-1 overflow-y-auto p-4 space-y-4">
        {messages.map((msg, idx) => (
          <div 
            key={idx} 
            // เพิ่ม Animation ข้อความใหม่
            className={`flex animate-in fade-in slide-in-from-bottom-2 duration-300 ${
              msg.role === "user" ? "justify-end" : "justify-start"
            }`}
          >
            <div className={`p-3 rounded-lg max-w-[80%] ${
              msg.role === "user" ? "bg-blue-600 text-white" : "bg-gray-100 dark:bg-gray-800"
            }`}>
              {msg.role === "assistant" ? (
                // ใช้ ReactMarkdown สำหรับข้อความ AI
                <ReactMarkdown className="prose prose-sm dark:prose-invert break-words">
                  {msg.content}
                </ReactMarkdown>
              ) : (
                <p className="whitespace-pre-wrap">{msg.content}</p>
              )}
            </div>
          </div>
        ))}

        {/* Typing Indicator แบบ Pulse */}
        {isTyping && (
          <div className="flex items-center gap-2 text-sm text-gray-500 animate-in fade-in duration-300">
            <span className="relative flex h-3 w-3">
              <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-blue-400 opacity-75"></span>
              <span className="relative inline-flex rounded-full h-3 w-3 bg-blue-500"></span>
            </span>
            Chann กำลังพิมพ์...
          </div>
        )}
      </div>

      {/* ช่อง Input */}
      <div className="p-4 border-t flex gap-2">
        <input 
          type="text" 
          value={input} 
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={(e) => e.key === "Enter" && sendMessage()}
          className="flex-1 border rounded px-3 py-2"
          placeholder="พิมพ์ข้อความ..."
        />
        <button onClick={sendMessage} className="bg-blue-600 text-white px-4 py-2 rounded">ส่ง</button>
      </div>
    </div>
  );
}
