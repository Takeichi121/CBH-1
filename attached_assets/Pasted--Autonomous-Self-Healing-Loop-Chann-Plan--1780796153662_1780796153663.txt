💻 โค้ดระบบ Autonomous & Self-Healing (ฉบับอัปเกรด)
เราจะสร้าง Loop ที่ให้ Chann วางแผน (Plan) และเลือกใช้เครื่องมือ หากพลาด Chann จะอ่าน Error แล้วแก้ไขตัวเองครับ

ไฟล์: server/services/autonomous-agent.ts

TypeScript
import { exec } from "child_process";
import { promisify } from "util";
import { getNBOSalesAuto } from "./nbo-service";
import { getAlohaSalesRaw } from "./aloha-service";
import { streamLLM } from "../replit_integrations/chat/services/llm-router";
import { pushToBoss } from "../socket";

const execAsync = promisify(exec);

export async function runChannTask(taskDescription: string) {
  let currentAttempt = 0;
  const maxRetries = 3;
  let lastError = "";

  while (currentAttempt < maxRetries) {
    console.log(`🤖 Chann กำลังวางแผนงานรอบที่ ${currentAttempt + 1}...`);

    // 1. ถาม AI ว่าจะใช้เครื่องมือไหนดี?
    let decision = "";
    await streamLLM({
      provider: "openai",
      mode: "code",
      message: `งานของนายคือ: ${taskDescription} 
                ข้อมูลที่นายมี: 
                - NBO (SQL) รหัส bk1040
                - Aloha (DBF) ตาม path ใน env
                หากรอบก่อนหน้าพลาด Error คือ: ${lastError}
                ตอบกลับเป็น JSON เท่านั้น: { "tool": "nbo" | "aloha" | "shell", "command": "string" }`,
      history: [],
      onToken: (t) => { decision += t; }
    });

    try {
      const plan = JSON.parse(decision.replace(/```json|```/g, ""));
      let result;

      // 2. ลงมือทำ (Execution)
      if (plan.tool === "nbo") {
        result = await getNBOSalesAuto(); // ใช้รหัส bk1040 อัตโนมัติ
      } else if (plan.tool === "aloha") {
        result = await getAlohaSalesRaw();
      } else if (plan.tool === "shell") {
        const { stdout } = await execAsync(plan.command, { timeout: 10000 });
        result = stdout;
      }

      // 3. รายงานผลให้นาย
      pushToBoss("chann-task-complete", {
        message: `✅ งานสำเร็จแล้วครับนาย: ${taskDescription}`,
        result: result
      });
      return result;

    } catch (error: any) {
      // 4. แก้ไขตัวเอง (Self-Healing)
      lastError = error.message;
      currentAttempt++;
      console.log(`❌ งานพลาด กำลังส่ง Error ให้สมองวิเคราะห์ใหม่...`);
    }
  }

  pushToBoss("chann-alert", {
    title: "⚠️ งานไม่สำเร็จ",
    message: `ผมพยายามแก้ปัญหาให้ 3 รอบแล้วแต่ยังติด Error: ${lastError} รบกวนนายช่วยดูหน่อยครับ`
  });
}
⚠️ ข้อควรระวังระดับวิกฤต (Security Warning)
ตามตารางตรวจสอบความพร้อม:

อันตรายจากการใช้ exec(): Chann อาจสร้างคำสั่งอันตราย เช่น rm -rf / ได้หากเกิดอาการหลอน (Hallucination)

ข้อแนะนำ: ควรจำกัดสิทธิ์ User ของ Node.js ไม่ให้เป็น root เพื่อป้องกันการลบไฟล์ระบบโดยไม่ตั้งใจ

การตั้งค่า: ตรวจสอบว่า .env ตั้งค่า ALOHA_DBF_PATH และรหัส SQL bk1040 ถูกต้องก่อนเริ่มรัน
